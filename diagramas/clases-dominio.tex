% Diagrama: Figura 10.x - Diagrama de clases del dominio del Gemelo Digital
% TikZ vectorial. Se incluye con \input desde contenido10.tex (página apaisada).
% Criterios de la dirección: solo se nombran los roles (no las asociaciones),
% se indica la navegabilidad y los atributos comunes se generalizan en clases
% abstractas (EntidadDigital y Perfil).
% Los roles del usuario se modelan como perfiles: un mismo Usuario puede tener
% a la vez PerfilConductor y PerfilPasajero sin cambiar de clase.
% Las cajas se ajustan a su contenido y se colocan de forma relativa; los
% carriles de las asociaciones se sitúan por debajo de la clase más alta.

\begin{tikzpicture}[x=1cm,y=1cm,
    every node/.append style={rectangle split part align={center,left,left}}]

  \def\sep{0.45}

  % ================= fila central: entidades del Gemelo Digital =================
  \node[clase, anchor=north west] (veh) at (0,0)
    {\textbf{Vehiculo}
     \nodepart{two} -matricula: String\\ -capacidad: int\\ -estado: EstadoVehiculo\\ -localizacion: Coordenada
     \nodepart{three} +actualizarDisponibilidad(): void};

  \node[clase, anchor=north west] (rut) at ($(veh.north east)+(\sep,0)$)
    {\textbf{Ruta}
     \nodepart{two} -distanciaKm: double\\ -duracionMin: int\\ -secuencia: List<Coordenada>
     \nodepart{three} +recalcular(): void};

  \node[clase, anchor=north west] (via) at ($(rut.north east)+(1.0,0)$)
    {\textbf{Viaje}
     \nodepart{two} -origen: Coordenada\\ -destino: Coordenada\\ -fechaHoraSalida: DateTime\\
       -plazasOfertadas: int\\ -plazasDisponibles: int\\ -desvioMaximo: double\\ -estado: EstadoViaje
     \nodepart{three} +publicar(): void\\ +iniciar(): void\\ +finalizar(): void\\ +cancelar(i: Incidencia): void};

  \node[clase, anchor=north west] (sol) at ($(via.north east)+(2.2,0)$)
    {\textbf{Solicitud}
     \nodepart{two} -fechaSolicitud: DateTime\\ -estado: EstadoSolicitud\\ -prioritaria: boolean
     \nodepart{three} +aceptar(): void\\ +rechazar(): void\\ +cancelar(): void};

  \node[clase, anchor=north west] (inc) at ($(sol.north east)+(1.3,0)$)
    {\textbf{Incidencia}
     \nodepart{two} -categoria: CategoriaIncidencia\\ -descripcion: String
     \nodepart{three} +notificarAdministrador(): void};

  \node[clase, anchor=north west] (usu) at ($(inc.north east)+(1.7,0)$)
    {\textbf{Usuario}
     \nodepart{two} -nombre: String\\ -email: String\\ -passwordHash: String\\ -toleranciaHoraria: Duration
     \nodepart{three} +registrarse(): void\\ +iniciarSesion(): Token\\ +actualizarDatos(): void\\
       +activarPerfil(tipo: String): Perfil};

  % ================= columna de perfiles (roles del usuario) =================
  \node[clase, anchor=north west] (per) at ($(usu.north east)+(1.8,0)$)
    {\textbf{\textit{Perfil}}\\ \textit{\{abstracta\}}
     \nodepart{two} \#activo: boolean\\ \#fechaAlta: DateTime
     \nodepart{three} +activar(): void\\ +desactivar(): void};

  % ================= fila superior: abstracta, evento y tipos =================
  \def\ytop{1.25}
  \node[clase, anchor=south] (ent) at ($(rut.north east)!0.5!(via.north west)+(0,\ytop)$)
    {\textbf{\textit{EntidadDigital}}\\ \textit{\{abstracta\}}
     \nodepart{two} \#id: UUID\\ \#creadoEn: DateTime\\ \#actualizadoEn: DateTime
     \nodepart{three} +registrarEvento(tipo: String): Evento};

  \node[enumeracion, anchor=south east] (estveh) at ($(ent.south west)+(-\sep,0)$)
    {\textit{\guillemotleft enumeración\guillemotright}\\ \textbf{EstadoVehiculo}
     \nodepart{two} DISPONIBLE\\ EN\_VIAJE\\ INACTIVO};

  \node[clase, anchor=south east] (coord) at ($(estveh.south west)+(-\sep,0)$)
    {\textbf{Coordenada}\\ \textit{\guillemotleft valor\guillemotright}
     \nodepart{two} -latitud: double\\ -longitud: double
     \nodepart{three} +distanciaA(otra: Coordenada): double};

  \node[clase, anchor=south west] (eve) at ($(ent.south east)+(1.6,0)$)
    {\textbf{Evento}
     \nodepart{two} -id: UUID\\ -tipo: String\\ -marcaTemporal: DateTime\\ -payload: JSON
     \nodepart{three} +serializar(): JSON};

  \node[enumeracion, anchor=south west] (estvia) at ($(eve.south east)+(\sep,0)$)
    {\textit{\guillemotleft enumeración\guillemotright}\\ \textbf{EstadoViaje}
     \nodepart{two} PUBLICADO\\ CONFIRMADO\\ EN\_CURSO\\ FINALIZADO\\ CANCELADO};

  \node[enumeracion, anchor=south west] (estsol) at ($(estvia.south east)+(\sep,0)$)
    {\textit{\guillemotleft enumeración\guillemotright}\\ \textbf{EstadoSolicitud}
     \nodepart{two} PENDIENTE\\ ACEPTADA\\ RECHAZADA\\ CANCELADA};

  \node[enumeracion, anchor=south west] (catinc) at ($(estsol.south east)+(\sep,0)$)
    {\textit{\guillemotleft enumeración\guillemotright}\\ \textbf{CategoriaIncidencia}
     \nodepart{two} TECNICA\\ CONDUCTOR\\ PASAJERO\\ OTRA};

  % Evento --> EntidadDigital: el evento conoce la entidad que lo origina
  \coordinate (ye) at ($(ent.south)+(0,0.55)$);
  \draw[asoc] (eve.west |- ye) -- (ent.east |- ye);
  \node[rol, above right] at (ent.east |- ye) {origen};
  \node[rol, below right] at (ent.east |- ye) {1};
  \node[rol, below left] at (eve.west |- ye) {0..*};

  % ---- generalización: las entidades del DT heredan de EntidadDigital ----
  \def\ybus{0.6}
  \draw[gen] (ent.south |- 0,\ybus) -- (ent.south);
  \draw[draw=Color2, line width=0.5pt] (veh.north |- 0,\ybus) -- (usu.north |- 0,\ybus);
  \foreach \n in {veh,via,sol,inc,usu}
    \draw[draw=Color2, line width=0.5pt] (\n.north |- 0,\ybus) -- (\n.north);

  % ---- asociaciones entre clases vecinas ----
  \def\yv{-1.3}
  % Viaje <>--> Ruta (composición)
  \draw[compos] (via.west |- 0,\yv) -- (rut.east |- 0,\yv);
  \node[rol, above right, xshift=0.12cm] at (rut.east |- 0,\yv) {ruta};
  \node[rol, below right, xshift=0.12cm] at (rut.east |- 0,\yv) {1};
  \node[rol, below left, xshift=-0.12cm] at (via.west |- 0,\yv) {1};
  % Viaje <--> Solicitud (bidireccional)
  \draw[asocbi] (via.east |- 0,\yv) -- (sol.west |- 0,\yv);
  \node[rol, above right, xshift=0.12cm] at (via.east |- 0,\yv) {viaje};
  \node[rol, below right, xshift=0.12cm] at (via.east |- 0,\yv) {1};
  \node[rol, above left, xshift=-0.12cm] at (sol.west |- 0,\yv) {solicitudes};
  \node[rol, below left, xshift=-0.12cm] at (sol.west |- 0,\yv) {0..*};
  % Incidencia --> Solicitud (solicitud afectada)
  \draw[asoc] (inc.west |- 0,\yv) -- (sol.east |- 0,\yv);
  \node[rol, above right, xshift=0.12cm] at (sol.east |- 0,\yv) {afectada};
  \node[rol, below right, xshift=0.12cm] at (sol.east |- 0,\yv) {0..1};
  \node[rol, below left, xshift=-0.12cm] at (inc.west |- 0,\yv) {0..1};
  % Incidencia --> Usuario (informante)
  \draw[asoc] (inc.east |- 0,\yv) -- (usu.west |- 0,\yv);
  \node[rol, above left, xshift=-0.12cm] at (usu.west |- 0,\yv) {informante};
  \node[rol, below left, xshift=-0.12cm] at (usu.west |- 0,\yv) {1};
  \node[rol, below right, xshift=0.12cm] at (inc.east |- 0,\yv) {0..*};
  % Usuario <>--> Perfil (composición: los perfiles son parte del usuario)
  \draw[compos] (usu.east |- 0,\yv) -- (per.west |- 0,\yv);
  \node[rol, above left, xshift=-0.12cm] at (per.west |- 0,\yv) {perfiles};
  \node[rol, below left, xshift=-0.12cm] at (per.west |- 0,\yv) {1..3};
  \node[rol, below right, xshift=0.12cm] at (usu.east |- 0,\yv) {1};

  % ---- carriles bajo la clase más alta ----
  \path let \p1=(veh.south), \p2=(via.south), \p3=(sol.south), \p4=(inc.south), \p5=(usu.south)
    in coordinate (lb) at (0,{min(\y1,\y2,\y3,\y4,\y5)});
  \coordinate (Ls) at ($(lb)+(0,-0.45)$);   % Solicitud --> PerfilPasajero
  \coordinate (Li) at ($(lb)+(0,-0.85)$);   % Incidencia --> Viaje
  \coordinate (Lw) at ($(lb)+(0,-1.35)$);   % Viaje --> Vehiculo
  \coordinate (Lv) at ($(lb)+(0,-1.85)$);   % Viaje --> PerfilConductor
  \coordinate (Le) at ($(lb)+(0,-2.25)$);   % Vehiculo --> PerfilConductor

  % perfiles concretos, alineados con los carriles que llegan a ellos
  \node[clase, anchor=west] (pas) at (per.west |- Ls)
    {\textbf{PerfilPasajero}
     \nodepart{two} -prioridadAlta: boolean
     \nodepart{three} +solicitarPlaza(v: Viaje): Solicitud};
  \node[clase, anchor=west] (con) at ($(per.west |- Lv)!0.5!(per.west |- Le)$)
    {\textbf{PerfilConductor}
     \nodepart{two} -declaracionLicencia: boolean\\ -fechaDeclaracion: DateTime
     \nodepart{three} +registrarVehiculo(): Vehiculo\\ +publicarViaje(): Viaje};
  \node[clase, anchor=north west] (adm) at ($(con.south west)+(0,-0.45)$)
    {\textbf{PerfilAdministrador}
     \nodepart{two} \mbox{}
     \nodepart{three} +gestionarPerfiles(u: Usuario): void\\ +lanzarSimulacion(e: Escenario): void\\
       +revisarIncidencia(i: Incidencia): void};

  % perfil abstracto intermedio: lo común a quien viaja (la valoración)
  \path let \p1=(pas.east), \p2=(con.east) in coordinate (t1) at ({max(\x1,\x2)+0.45},0);
  \coordinate (mv) at ($(pas.east)!0.5!(con.east)$);
  \node[clase, anchor=west] (vij) at ($(t1 |- mv)+(0.45,0)$)
    {\textbf{\textit{PerfilViajero}}\\ \textit{\{abstracta\}}
     \nodepart{two} \#valoracionMedia: double
     \nodepart{three} +recibirValoracion(p: int): void};

  % generalización: PerfilConductor y PerfilPasajero --|> PerfilViajero
  \draw[gen] (t1 |- vij.west) -- (vij.west);
  \draw[draw=Color2, line width=0.5pt] (t1 |- pas.east) -- (t1 |- con.east);
  \foreach \n in {pas,con}
    \draw[draw=Color2, line width=0.5pt] (t1 |- \n.east) -- (\n.east);

  % generalización: PerfilViajero y PerfilAdministrador --|> Perfil (tronco a la derecha)
  \path let \p1=(per.east), \p2=(vij.east) in coordinate (tr) at ({max(\x1,\x2)+0.4},0);
  \draw[gen] (tr |- per.east) -- (per.east);
  \draw[draw=Color2, line width=0.5pt] (tr |- per.east) -- (tr |- adm.east);
  \foreach \n in {vij,adm}
    \draw[draw=Color2, line width=0.5pt] (tr |- \n.east) -- (\n.east);

  % Solicitud --> PerfilPasajero (pasajero)
  \coordinate (s1) at ($(sol.south)+(0.2,0)$);
  \draw[asoc] (s1) -- (s1 |- Ls) -- (pas.west |- Ls);
  \node[rol, above left] at (pas.west |- Ls) {pasajero\ 1};
  \node[rol, right] at ($(s1)+(0,-0.2)$) {0..*};
  % Incidencia --> Viaje
  \coordinate (s2) at ($(inc.south)+(-0.3,0)$);  \coordinate (t2) at ($(via.south)+(0.45,0)$);
  \draw[asoc] (s2) -- (s2 |- Li) -- (t2 |- Li) -- (t2);
  \node[rol, right] at ($(t2)+(0,-0.2)$) {viaje\ 1};
  \node[rol, left]  at ($(s2)+(0,-0.2)$) {0..*};
  % Viaje --> Vehiculo (vehículo con el que se realiza el viaje)
  \coordinate (s5) at ($(via.south)+(-0.75,0)$);  \coordinate (t5) at ($(veh.south)+(0.6,0)$);
  \draw[asoc] (s5) -- (s5 |- Lw) -- (t5 |- Lw) -- (t5);
  \node[rol, right] at ($(t5)+(0,-0.2)$) {vehiculo\ 1};
  \node[rol, left]  at ($(s5)+(0,-0.2)$) {0..*};
  % Viaje --> PerfilConductor (conductor)
  \coordinate (s3) at ($(via.south)+(-0.25,0)$);
  \draw[asoc] (s3) -- (s3 |- Lv) -- (con.west |- Lv);
  \node[rol, above left] at (con.west |- Lv) {conductor\ 1};
  \node[rol, left] at ($(s3)+(0,-0.2)$) {0..*};
  % Vehiculo --> PerfilConductor (propietario)
  \coordinate (s4) at (veh.south);
  \draw[asoc] (s4) -- (s4 |- Le) -- (con.west |- Le);
  \node[rol, below left] at (con.west |- Le) {propietario\ 1};
  \node[rol, right] at ($(s4)+(0,-0.2)$) {0..*};

  % ================= fila inferior: simulación =================
  \coordinate (rc) at ($(Le)+(0,-0.6)$);
  \node[clase, anchor=north west] (res) at (veh.west |- rc)
    {\textbf{ResultadoSimulacion}
     \nodepart{two} -id: UUID\\ -semilla: long\\ -vehiculosUtilizados: int\\ -vehiculosEvitados: int\\
       -ocupacionMedia: double\\ -solicitudesCubiertas: double\\ -tiempoRespuestaMatching: Duration
     \nodepart{three} +compararCon(otro: ResultadoSimulacion): Informe};

  \node[clase, anchor=north west] (esc) at ($(res.north east)+(1.6,0)$)
    {\textbf{Escenario}
     \nodepart{two} -id: UUID\\ -nombre: String\\ -numUsuarios: int\\ -nivelAdopcion: double\\
       -repartoModal: double\\ -ventanaTemporal: Duration\\ -capacidadMedia: int\\ -replicas: int
     \nodepart{three} +ejecutar(): List<ResultadoSimulacion>};

  \node[clase, anchor=north west] (age) at ($(esc.north east)+(1.6,0)$)
    {\textbf{AgenteSimulado}
     \nodepart{two} -origen: Coordenada\\ -destino: Coordenada\\ -horaDeseada: DateTime\\
       -modoHabitual: ModoTransporte\\ -conduce: boolean
     \nodepart{three} +decidirModo(): ModoTransporte};

  \node[enumeracion, anchor=north west] (modo) at ($(age.north east)+(\sep,0)$)
    {\textit{\guillemotleft enumeración\guillemotright}\\ \textbf{ModoTransporte}
     \nodepart{two} COCHE\_PROPIO\\ TRANSPORTE\_PUBLICO\\ OTRO};

  % Escenario <>--> ResultadoSimulacion (un resultado por réplica)
  \coordinate (yr) at ($(esc.north)+(0,-1.2)$);
  \draw[compos] (esc.west |- yr) -- (res.east |- yr);
  \node[rol, above right] at (res.east |- yr) {resultados};
  \node[rol, below right] at (res.east |- yr) {1..*};
  \node[rol, below left] at (esc.west |- yr) {1};
  % Escenario <>--> AgenteSimulado (población sintética)
  \draw[compos] (esc.east |- yr) -- (age.west |- yr);
  \node[rol, above left] at (age.west |- yr) {agentes};
  \node[rol, below left] at (age.west |- yr) {1..*};
  \node[rol, below right] at (esc.east |- yr) {1};
\end{tikzpicture}
