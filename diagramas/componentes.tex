% Diagrama: Figura B.2 - Componentes del backend
% TikZ vectorial. Se incluye con \input desde apendices.tex.
% Fila superior: servicios de dominio expuestos por el API Gateway.
% Fila inferior: componentes que esos servicios usan (matching, DT,
% notificaciones) y la simulación, que se lanza desde el panel de administración.

\begin{tikzpicture}[x=1cm,y=1cm]
  \useasboundingbox (0.0,0.95) rectangle (15.4,8.35);
  \tikzset{bus/.style={draw=Color2!75, line width=0.45pt, dashed}}

  % ---------------- frontera ----------------
  \draw[draw=Color2!55, dashed, rounded corners=4pt, fill=Color2!2]
    (0.60,1.55) rectangle (11.85,7.55);
  \node[font=\scriptsize\bfseries, text=Color2, anchor=south west] at (0.75,7.60)
    {\textit{Backend} (Node.js + NestJS)};

  % ---------------- componentes ----------------
  \node[componente, minimum width=2.90cm] (gate) at (6.275,6.75) {API Gateway};

  \node[componente, minimum width=2.30cm] (usu) at (2.30,5.00) {Usuarios};
  \node[componente, minimum width=2.30cm] (sol) at (4.95,5.00) {Solicitudes};
  \node[componente, minimum width=2.30cm] (inc) at (7.60,5.00) {Incidencias};
  \node[componente, minimum width=2.30cm] (via) at (10.25,5.00) {Viajes};

  \node[componente, minimum width=2.30cm] (sim) at (2.30,3.55) {Simulación};
  \node[componente, minimum width=2.30cm] (dt)  at (4.95,3.55) {Gemelo Digital};
  \node[componente, minimum width=2.30cm] (not) at (7.60,3.55) {Notificaciones};
  \node[componente, minimum width=2.30cm] (mat) at (10.25,3.55) {\textit{Matching}};

  \node[componente, minimum width=10.50cm] (repo) at (6.40,2.20) {Acceso a datos};

  % ---------------- elementos externos ----------------
  \node[componente, minimum width=2.50cm, fill=white] (fe)   at (13.45,6.75) {\textit{Frontend} y panel};
  \node[componente, minimum width=2.50cm, fill=white] (osrm) at (13.45,3.55) {Motor de rutas\\ (OSRM)};
  \node[componente, minimum width=2.50cm, fill=white] (bd)   at (13.45,2.20) {PostgreSQL + PostGIS};

  % ---------------- dependencias ----------------
  \draw[dep] (fe.west) -- (gate.east);
  % API Gateway -> servicios de dominio
  \draw[bus] (gate.south) -- (6.275,5.90);
  \draw[bus] (2.30,5.90) -- (10.25,5.90);
  \foreach \c in {usu,sol,inc,via} \draw[dep] (\c.north |- 0,5.90) -- (\c.north);
  % API Gateway -> Simulación (lanzada desde el panel de administración)
  \draw[dep] (gate.west) -- (0.85,6.75) -- (0.85,3.55) -- (sim.west);

  % servicios -> componentes de apoyo
  \draw[dep] (sol.south) -- (dt.north);
  \draw[dep] (inc.south) -- (not.north);
  \draw[dep] (via.south) -- (mat.north);
  \draw[dep] (sim.east) -- (dt.west);
  \draw[dep] (mat.east) -- (osrm.west);

  % persistencia
  \foreach \c in {usu,sol,inc,via} {
    \coordinate (l) at ($(\c.south)+(1.325,-0.25)$);
    \draw[dep] ($(\c.south)+(0.85,0)$) -- ($(\c.south)+(0.85,-0.25)$) -- (l) -- (l |- repo.north);
  }
  \foreach \c in {sim,dt,mat} \draw[dep] (\c.south) -- (\c.south |- repo.north);
  \draw[dep] (repo.east) -- (bd.west);
\end{tikzpicture}
