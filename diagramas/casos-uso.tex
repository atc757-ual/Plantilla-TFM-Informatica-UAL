% Diagrama: Figura 7.1 - Diagrama de casos de uso
% TikZ vectorial. Se incluye con \input desde contenido7.tex.
% - Jerarquía de actores paralela a la de perfiles del diagrama de clases:
%   Usuario (abstracto: acceso y perfil) <- Viajero (abstracto: viaje y ruta)
%   <- Conductor / Pasajero; Administrador especializa directamente a Usuario.
% - El Gemelo Digital es parte del sistema: no es actor ni caso de uso.
% - <<extend>> va del caso que extiende al caso base: CU-05 extiende la
%   búsqueda (CU-04) y CU-11 extiende la gestión de estados (CU-07).

\begin{tikzpicture}[x=1cm,y=1cm,
    cu/.style={ellipse, draw=Color2, fill=Color2!8, line width=0.5pt, font=\scriptsize,
      align=center, minimum width=3.2cm, minimum height=0.72cm, inner sep=0.5pt},
    lazo/.style={draw=Color2, line width=0.5pt},
    incl/.style={draw=Color2, line width=0.5pt, dashed, -{Straight Barb[length=1.6mm,width=2.2mm]}},
    estereo/.style={font=\tiny\itshape, text=Color2!90!black, fill=white, inner sep=1pt},
    actor/.pic={
      \draw[Color2, line width=0.6pt] (0,0.55) circle (0.16);
      \draw[Color2, line width=0.6pt] (0,0.39) -- (0,-0.05)
        (-0.28,0.25) -- (0.28,0.25) (0,-0.05) -- (-0.22,-0.42) (0,-0.05) -- (0.22,-0.42);
    }]

  % ---------------- límite del sistema ----------------
  \draw[draw=Color2!70, line width=0.5pt, rounded corners=2pt] (2.75,1.9) rectangle (13.95,13.45);
  \node[font=\footnotesize\bfseries, text=Color2, anchor=north west] at (2.85,13.4)
    {Plataforma de carpooling universitario P2P};

  % ---------------- actores ----------------
  \pic at (1.0,13.3) {actor};    \node[font=\scriptsize\itshape, left] at (0.7,13.3) {Usuario};
  \pic at (1.0,10.85) {actor};   \node[font=\scriptsize\itshape, below] at (1.0,10.4) {Viajero};
  \pic at (1.65,6.6) {actor};    \node[font=\scriptsize, below] at (1.65,6.15) {Conductor};
  \pic at (0.55,3.0) {actor};    \node[font=\scriptsize, below] at (0.55,2.55) {Pasajero};
  \pic at (15.15,11.35) {actor}; \node[font=\scriptsize, below] at (15.4,10.9) {Administrador};

  % generalización de actores:
  % Viajero y Administrador son Usuarios; Conductor y Pasajero son Viajeros
  \draw[gen] (1.0,11.58) -- (1.0,12.86);
  \draw[gen] (15.15,12.1) -- (15.15,14.45) -- (1.0,14.45) -- (1.0,14.03);
  \draw[gen] (1.62,7.32) -- (1.3,9.98);
  \draw[gen] (0.58,3.72) -- (0.72,9.98);

  % ---------------- casos de uso ----------------
  % comunes (Usuario)
  \node[cu] (cu01) at (4.85,12.55) {CU-01\\ Registrarse / iniciar sesión};
  \node[cu] (cu02) at (4.85,11.45) {CU-02\\ Gestionar perfil};
  \node[cu] (cu08) at (4.85,10.35) {CU-08\\ Consultar ruta / ubicación};
  % conductor
  \node[cu] (cu03) at (4.85,7.4)  {CU-03\\ Publicar viaje};
  \node[cu] (cu06) at (4.85,6.25) {CU-06\\ Aceptar / rechazar pasajero};
  % pasajero: busca y, desde los resultados, solicita plaza
  \node[cu] (cu04) at (4.85,3.0)  {CU-04\\ Buscar viajes};
  \node[cu] (cu05) at (9.3,3.0)   {CU-05\\ Solicitar plaza};
  % núcleo de estados
  \node[cu] (cu07) at (9.3,6.85)  {CU-07\\ Gestionar viaje\\ (estados)};
  \node[cu] (cu11) at (12.2,4.9)  {CU-11\\ Notificar cambios};
  % administración
  \node[cu] (cu13) at (11.95,11.65) {CU-13\\ Gestionar usuarios y perfiles};
  \node[cu] (cu09) at (11.95,10.45) {CU-09\\ Simular escenario};
  \node[cu] (cu10) at (11.95,9.25)  {CU-10\\ Consultar métricas};
  \node[cu] (cu12) at (11.95,8.1)   {CU-12\\ Revisar incidencias};

  % ---------------- asociaciones actor -- caso de uso ----------------
  \foreach \c in {cu01,cu02} \draw[lazo] (1.35,13.4) -- (\c.west);
  \draw[lazo] (1.35,10.95) -- (cu08.west);
  \draw[lazo] (1.35,10.75) -- (cu07.north);
  \foreach \c in {cu03,cu06} \draw[lazo] (1.95,6.7) -- (\c.west);
  \draw[lazo] (0.85,3.05) -- (cu04.west);
  \foreach \c in {cu13,cu09,cu10,cu12} \draw[lazo] (14.8,11.35) -- (\c.east);

  % ---------------- include / extend ----------------
  \draw[incl] (cu03.east) -- node[estereo, pos=0.45, above, sloped] {\guillemotleft include\guillemotright} (cu07.west);
  \draw[incl] (cu06.east) -- node[estereo, pos=0.45, below, sloped] {\guillemotleft include\guillemotright} (cu07.south west);
  \draw[incl] (cu05.north) -- node[estereo, pos=0.5, right] {\guillemotleft include\guillemotright} (cu07.south);
  \draw[incl] (cu05.west) -- node[estereo, above] {\guillemotleft extend\guillemotright} (cu04.east);
  \draw[incl] (cu11.north west) -- node[estereo, pos=0.5, above, sloped] {\guillemotleft extend\guillemotright} (cu07.east);
\end{tikzpicture}
